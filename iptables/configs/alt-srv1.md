# /etc/sysconfig/iptables

:PREROUTING ACCEPT [670:61831]

-A POSTROUTING -s 192.168.100.0/24 -o ens33 -j MASQUERADE
-A POSTROUTING -s 192.168.200.0/24 -o ens33 -j MASQUERADE

*filter
:INPUT ACCEPT [160547:78435507]
